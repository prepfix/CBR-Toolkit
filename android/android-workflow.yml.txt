name: Build Android APK

on:
  workflow_dispatch:
  push:
    paths: ["android/**", "index.html", "web.json", "words.txt"]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - name: Put the reading page inside the app
        run: |
          mkdir -p android/app/src/main/assets
          cp index.html web.json words.txt android/app/src/main/assets/
      - name: Accept Android SDK licenses
        run: yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: 8.9
      - name: Build debug APK
        run: gradle -p android --no-daemon assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: CBR-Toolkit-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
